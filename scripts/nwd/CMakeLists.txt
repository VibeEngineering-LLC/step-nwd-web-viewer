cmake_minimum_required(VERSION 3.20)
project(nwd_tools CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# NWD_READER_DIR: путь к локальному клону nwd-reader (проверенному вами; в репозиторий не входит)
add_subdirectory(${NWD_READER_DIR} nwd-reader EXCLUDE_FROM_ALL)
foreach(t nwd_dump nwd_probe)
  add_executable(${t} ${t}.cpp)
  target_link_libraries(${t} PRIVATE nwd::reader)
  if(MSVC)
    target_compile_options(${t} PRIVATE /utf-8 /EHsc)
  endif()
endforeach()
